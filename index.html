Vercel-də DEEPSEEK_API_KEY-i və Redeploy vəziyyətini yoxlayın."); } if (!res.ok) throw new Error(data.error || "Server xətası baş verdi"); const musbixCode = data.result; console.log("DeepSeek-dən gələn xam cavab:\n", musbixCode); const charCount = musbixCode.length; const tokenCost = Math.ceil(charCount / 100) * 10; if (currentJeton < tokenCost) { throw new Error(`Kifayət qədər jetonunuz yoxdur. Bu musiqi üçün ${tokenCost} jeton lazımdır.`); } currentJeton = Math.max(0, currentJeton - tokenCost); if (currentUser) { localStorage.setItem("musbix_jeton_" + currentUser.uid, currentJeton); } els.jNum.innerText = `${currentJeton} Jeton`; // YENİ: Bütün mətndən (sətir sərhədlərinə baxmadan) kodları qlobal şəkildə "ovlayırıq". // Artıq DeepSeek arada "Bölüm 1", zaman izahatı və ya başqa mətn yazsa belə, // bu, kodun tapılmasına mane olmur. const codeRegex = /([A-Z]{2})\s*:\s*([A-Za-z0-9#]+)\s*-\s*([0-9.]+)/g; const matches = [...musbixCode.matchAll(codeRegex)]; const playData = []; const usedInstruments = new Set(); matches.forEach(([, inst, note, time]) => { playData.push({ inst, note, time: parseFloat(time) }); usedInstruments.add(inst); }); if (playData.length === 0) { console.warn("Heç bir düzgün kod tapılmadı. Xam cavabı yuxarıdan yoxlayın."); throw new Error("AI düzgün formatda kod qaytarmadı.
